class Solution {
    List<List<String>> output ;
    public List<List<String>> partition(String s) {
        this.output = new ArrayList<>();
        List<String> palindrome = new ArrayList<>();
        helper(s, 0, palindrome); 
        return output;
    }
    private void helper(String s, int idx, List<String> palindrome){
        if( idx == s.length()) {
            output.add(new ArrayList<>(palindrome));
            return ;
        }
        StringBuilder sb = new StringBuilder();
        for(int i = idx; i<s.length() ; i++){
            sb.append(s.charAt(i));// chop the string
            String sub = sb.toString();
            if(isPalindrome(sub)) {
                palindrome.add(sub);
                helper(s,i+1,palindrome);
                palindrome.removeLast();
            }//check for palindrome  
            // if true add and proceede with the remaining half 
        }
    }
    private boolean isPalindrome(String sb){
        int i = 0;
        int j = sb.length()-1;
        while(i <= j){
            if(sb.charAt(i) != sb.charAt(j)) return false;
            i++;
            j--;
        }
        return true;
    }
}
